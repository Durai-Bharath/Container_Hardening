docker logs tracee-nginx > /tmp/nginx_trace_raw.log 2>&1

jq -c --arg id "$NGINX_ID" '
  select(
    type == "object" and
    .syscall and
    (
      .containerId == $id or
      .container.id == $id
    )
  )
  | {
      timestamp: ((.timestamp // .ts) / 1000000000),
      syscall: .syscall,
      pid: (.processId // .pid),
      process: (.processName // .comm)
    }
' /tmp/nginx_trace_raw.log > traces/nginx_discovery_trace.jsonl

wc -l traces/nginx_discovery_trace.jsonl
head -n 3 traces/nginx_discovery_trace.jsonl
