docker rm -f tracee-mysql mysql-discovery 2>/dev/null || true

docker run -d \
  --name tracee-mysql \
  --privileged \
  --pid=host \
  --cgroupns=host \
  -v /var/run/docker.sock:/var/run/docker.sock \
  -v /etc/os-release:/etc/os-release-host:ro \
  -v /lib/modules:/lib/modules:ro \
  -v /usr/src:/usr/src:ro \
  -e LIBBPFGO_OSRELEASE_FILE=/etc/os-release-host \
  aquasec/tracee:latest \
  --scope container=new \
  --events syscalls \
  --output json

docker logs tracee-mysql > /tmp/mysql_trace_raw.log 2>&1
docker rm -f tracee-mysql


jq -c --arg id "$MYSQL_ID" '
  select(
    type == "object" and
    .syscall and
    (
      .containerId == $id or
      .container.id == $id
    )
  )
  | {
      timestamp: ((.timestamp // .ts) / 1000000000),
      syscall: .syscall,
      pid: (.processId // .pid),
      process: (.processName // .comm)
    }
' /tmp/mysql_trace_raw.log > traces/mysql_discovery_trace.jsonl

wc -l traces/mysql_discovery_trace.jsonl
head -n 3 traces/mysql_discovery_trace.jsonl
