  --name tracee-redis \
  --privileged \
  --pid=host \
  --cgroupns=host \
  -v /var/run/docker.sock:/var/run/docker.sock \
  -v /etc/os-release:/etc/os-release-host:ro \
  -v /lib/modules:/lib/modules:ro \
  -v /usr/src:/usr/src:ro \
  -e LIBBPFGO_OSRELEASE_FILE=/etc/os-release-host \
  aquasec/tracee:latest \
  --scope container=new \
  --events syscalls \
  --output json


  docker run -d \
  --name redis-discovery \
  --network host \
  redis:7 \
  redis-server --save "" --appendonly no --protected-mode no


  timeout 30 redis-benchmark \
  -h 127.0.0.1 \
  -p 6379 \
  -c 10 \
  -n 100000 \
  -t get,set,incr,lpush,lpop


  REDIS_ID=$(docker inspect -f '{{.Id}}' redis-discovery)
  docker rm -f redis-discovery

docker logs tracee-redis > /tmp/redis_trace_raw.log 2>&1
docker rm -f tracee-redis


jq -c --arg id "$REDIS_ID" '
  select(
    type == "object" and
    .syscall and
    (
      .containerId == $id or
      .container.id == $id
    )
  )
  | {
      timestamp: ((.timestamp // .ts) / 1000000000),
      syscall: .syscall,
      pid: (.processId // .pid),
      process: (.processName // .comm)
    }
' /tmp/redis_trace_raw.log > traces/redis_discovery_trace.jsonl


wc -l traces/redis_discovery_trace.jsonl
ls -lh traces/redis_discovery_trace.jsonl
head -n 3 traces/redis_discovery_trace.jsonl
